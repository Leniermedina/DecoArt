// Modal WhatsApp (similar) const modal = document.getElementById('whatsapp-modal'); const floatingWhatsApp = document.getElementById('whatsapp-floating'); const closeModal = document.querySelector('.close-modal'); floatingWhatsApp.addEventListener('click', () => { modal.style.display = 'flex'; }); closeModal.addEventListener('click', () => { modal.style.display = 'none'; }); window.addEventListener('click', (e) => { if (e.target === modal) { modal.style.display = 'none'; } }); document.getElementById('whatsapp-form').addEventListener('submit', async (e) => { e.preventDefault(); const nombre = document.getElementById('nombre').value.trim(); const apellidos = document.getElementById('apellidos').value.trim(); const telefono = document.getElementById('telefono').value.trim(); const direccion = document.getElementById('direccion').value.trim(); if (!nombre || !apellidos || !telefono || !direccion) { alert('Por favor complete todos los campos'); return; } const cart = getCart(); if (cart.length === 0) { alert('El carrito está vacío'); return; } try { // Cargar productos.json (aunque tengas productosData, mejor asegurar) const response = await fetch('productos.json'); const data = await response.json(); const productos = data.categorias.flatMap(cat => cat.productos); let mensajeWhatsApp = `Buenos días, quiero hacer este pedido:\n`; mensajeWhatsApp += `Mi nombre es ${nombre} ${apellidos}.\n`; mensajeWhatsApp += `Mi número es ${telefono}.\n`; mensajeWhatsApp += `Mi dirección es ${direccion}.\n\n`; mensajeWhatsApp += `Pedido:\n`; const pedidoDetalle = []; let total = 0; cart.forEach(item => { const prod = productos.find(p => p.id === item.id); if (prod) { const nombreProd = currentLanguage === 'es' ? prod.nombre_es : prod.nombre_en; const subtotal = prod.precio * item.cantidad; mensajeWhatsApp += `- ${item.cantidad} x ${nombreProd} (${CONFIG.CURRENCY}${subtotal.toFixed(2)})\n`; total += subtotal; pedidoDetalle.push({ cantidad: item.cantidad, nombre: nombreProd, precio: `${CONFIG.CURRENCY}${prod.precio.toFixed(2)}`, subtotal: `${CONFIG.CURRENCY}${subtotal.toFixed(2)}` }); } }); mensajeWhatsApp += `Total: ${CONFIG.CURRENCY}${total.toFixed(2)}`; const totalFormateado = `${CONFIG.CURRENCY}${total.toFixed(2)}`; const urlWhatsApp = `https://wa.me/${CONFIG.WHATSAPP_NUMBER}?text=${encodeURIComponent(mensajeWhatsApp)}`; window.open(urlWhatsApp, '_blank'); const backendUrl = 'https://tu-backend.onrender.com/api/crear-gist-pedido'; // CAMBIA ESTO const pedidoBackend = { nombre, apellidos, telefono, direccion, pedido: pedidoDetalle, total: totalFormateado }; const resBackend = await fetch(backendUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(pedidoBackend) }); const result = await resBackend.json(); if (result.success) { console.log('✅ Pedido guardado en GitHub:', result.gistUrl); } else { console.error('❌ Error del backend:', result.error); } } catch (error) { console.error('Error al procesar el pedido:', error); } finally { modal.style.display = 'none'; document.getElementById('whatsapp-form').reset(); } });
